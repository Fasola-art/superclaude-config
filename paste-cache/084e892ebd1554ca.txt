  신호 통합 로직                                                                                                                                               
                                                                                                                                                               
  # 예시: 통합 트레이딩 신호                                                                                                                                   
  def calculate_signal():                                                                                                                                      
      # Jetson: 뉴스 감정 (0~1)                                                                                                                                
      news_sentiment = get_finbert_score()  # 0.7                                                                                                              
                                                                                                                                                               
      # 라즈베리: 차트 패턴 (0~1)                                                                                                                              
      chart_pattern = get_yolo_pattern()    # 0.8                                                                                                              
                                                                                                                                                               
      # 4090: 기술적 지표                                                                                                                                      
      technical = get_indicators()          # 0.6                                                                                                              
                                                                                                                                                               
      # 가중치 적용                                                                                                                                            
      signal = (                                                                                                                                               
          news_sentiment * 0.3 +                                                                                                                               
          chart_pattern * 0.4 +                                                                                                                                
          technical * 0.3                                                                                                                                      
      )                                                                                                                                                        
                                                                                                                                                               
      return signal  # > 0.7 이면 매수
